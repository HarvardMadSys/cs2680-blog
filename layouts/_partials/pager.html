{{/* Pagination for a list of posts. Takes the page, whose .Paginator the template has already created with .Paginate. The number row is windowed -- first, last, and the current page with one either side -- so a tag with thirty pages does not print thirty links. The whole nav is hidden while the filter box has a query in it, because a filter searches every page and a pager through the unfiltered list would then be pointing somewhere else entirely. */}} {{- $p := .Paginator }} {{- if gt $p.TotalPages 1 }} {{- $cur := $p.PageNumber }} {{- $last := $p.TotalPages }}
{{- with $p.Prev }} ← {{- else }} ← {{- end }} {{- $win := slice }} {{- range $p.Pagers }} {{- $n := .PageNumber }} {{- if or (eq $n 1) (eq $n $last) (and (ge $n (sub $cur 1)) (le $n (add $cur 1))) }} {{- $win = $win | append $n }} {{- end }} {{- end }} {{- $prev := 0 }} {{- range $win }} {{- if and (gt $prev 0) (gt (sub . $prev) 1) }}…{{ end }} {{- if eq . $cur }} {{ . }} {{- else }} {{ . }} {{- end }} {{- $prev = . }} {{- end }} {{- with $p.Next }} → {{- else }} → {{- end }}
Page {{ $cur }} of {{ $last }}
{{- end }}
